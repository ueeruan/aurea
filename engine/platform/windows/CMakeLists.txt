if(NOT TARGET aurea_vulkan)
    message(FATAL_ERROR "AUREA Desktop requires Vulkan headers at build time and a Vulkan driver at runtime")
endif()
add_library(aurea_windows_media STATIC WindowsMedia.cpp)
target_link_libraries(aurea_windows_media PUBLIC aurea_core mfplat mfreadwrite mfuuid ole32 propsys windowscodecs winmm)
target_compile_definitions(aurea_windows_media PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(aurea_desktop WIN32 Desktop.cpp DesktopTests.cpp Aurea.rc)
set_target_properties(aurea_desktop PROPERTIES OUTPUT_NAME "AUREA")
target_link_libraries(aurea_desktop PRIVATE aurea_windows_media aurea_vulkan gdiplus comctl32 comdlg32 shell32 dwmapi)
target_compile_options(aurea_desktop PRIVATE /utf-8)
target_link_options(aurea_desktop PRIVATE /MANIFEST:NO)
target_compile_options(aurea_windows_media PRIVATE /utf-8)
target_compile_definitions(aurea_desktop PRIVATE _CRT_SECURE_NO_WARNINGS)
install(TARGETS aurea_desktop RUNTIME DESTINATION .)
